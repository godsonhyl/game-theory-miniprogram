## 1. 建表：poker_card_grants（发放/扣减流水）

```sql
CREATE TABLE poker_card_grants (
  id            bigserial    PRIMARY KEY,
  owner_id      uuid         NOT NULL DEFAULT auth.uid(),  -- 同 profiles：客户端不传，服务端填
  kind          text         NOT NULL,   -- 'daily' | 'hands' | 'consume' | 'refund' | 'purchase'
  amount        int          NOT NULL,   -- 正=发放，负=扣减
  balance_after int          NOT NULL,   -- 出账即审计，事后对账不再翻全表
  day           date,                    -- kind='daily' 时记，幂等键的一半
  hand_base     int,                    -- kind='hands' 时记推进后的 bonus_base
  token         text,                    -- 幂等凭据（如建房 room_no）；NULL 表示不参与幂等
  created_at    timestamptz  NOT NULL DEFAULT now()
);

-- 幂等键 1：签到（owner_id, day, kind='daily'）唯一，同一天同人不重复发放
CREATE UNIQUE INDEX poker_card_grants_daily_uidx
  ON poker_card_grants (owner_id, day)
  WHERE kind = 'daily';

-- 幂等键 2：扣卡凭据（owner_id, token）唯一，同 token 不重复扣减
CREATE UNIQUE INDEX poker_card_grants_token_uidx
  ON poker_card_grants (owner_id, token)
  WHERE token IS NOT NULL;
```

## 2. 列权限收口：poker_profiles 关键列客户端只读

```sql
-- 撤销客户端（anon/authenticated 角色）对关键列的写权限
-- WorkBuddy 的 RLS + 列权限模型下，客户端只能写 nickname/avatar
REVOKE UPDATE (room_cards, total_hands, total_wins, total_net, bonus_base, last_checkin)
  ON poker_profiles FROM anon, authenticated;

-- 客户端 update 白名单只剩 nickname/avatar（其余列显式保留只读）
-- 注：若 WorkBuddy 控制台以列权限 UI 管理，则在此处勾选「客户端可写列 =
-- nickname, avatar」即可，SQL 形式如上 REVOKE。

-- poker_card_grants：客户端完全不可直接写（写入全经 rpc SECURITY DEFINER 函数）
REVOKE INSERT, UPDATE, DELETE ON poker_card_grants FROM anon, authenticated;
GRANT SELECT ON poker_card_grants TO anon, authenticated;  -- 只读（审计/展示用）
```

## 3. 服务端函数（4 个 PL/pgSQL SECURITY DEFINER）

### 3.1 poker_claim_daily(p_day date) —— 签到（幂等）

```sql
CREATE OR REPLACE FUNCTION poker_claim_daily(p_day date)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_uid   uuid   := auth.uid();
  v_cards int    := 2;          -- 每日发放张数（与客户端 CARD_RULES.daily 一致）
  v_bal   int;
  v_day   date  := COALESCE(p_day, current_date);  -- 服务端定日期，不信客户端
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION '未登录' USING ERRCODE = '42501';
  END IF;

  -- 幂等：先插流水，唯一约束 (owner_id, day WHERE kind='daily') 冲突即今日已领
  BEGIN
    INSERT INTO poker_card_grants (owner_id, kind, amount, balance_after, day)
    VALUES (v_uid, 'daily', v_cards, 0, v_day);
  EXCEPTION WHEN unique_violation THEN
    -- 今日已领：返回当前余额（不重复发）
    SELECT room_cards INTO v_bal FROM poker_profiles WHERE owner_id = v_uid;
    RETURN jsonb_build_object('ok', false, 'reason', '今日已领取，明天再来',
                              'roomCards', v_bal);
  END;

  -- 插流水成功 → 同事务发卡 + 置 last_checkin
  UPDATE poker_profiles
     SET room_cards   = room_cards + v_cards,
         last_checkin = v_day
   WHERE owner_id = v_uid
  RETURNING room_cards INTO v_bal;

  -- 回写流水的 balance_after（同事务内对账）
  UPDATE poker_card_grants
     SET balance_after = v_bal
   WHERE owner_id = v_uid AND kind = 'daily' AND day = v_day
   ORDER BY id DESC LIMIT 1;

  RETURN jsonb_build_object('ok', true, 'gained', v_cards, 'roomCards', v_bal);
END;
$$;
```

### 3.2 poker_claim_hands() —— 对局奖励（拉档抢占）

```sql
CREATE OR REPLACE FUNCTION poker_claim_hands()
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_uid   uuid  := auth.uid();
  v_hands int;
  v_base  int;
  v_cap   int   := 5;          -- 单次上限（与客户端 CARD_RULES.bonusCap 一致）
  v_per   int   := 10;         -- 每多少手 1 张（与 CARD_RULES.handsPerCard 一致）
  v_gained int;
  v_bal   int;
  v_new_base int;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION '未登录' USING ERRCODE = '42501';
  END IF;

  -- 一条语句并发抢占：读 total_hands/bonus_base → 算可领 → 推进基准 → 返回
  -- 多个并发请求只有一个能命中（其余基准已推进，gained=0）
  SELECT total_hands, bonus_base INTO v_hands, v_base
  FROM poker_profiles WHERE owner_id = v_uid FOR UPDATE;

  v_gained := LEAST(v_cap, FLOOR((v_hands - v_base) / v_per));
  IF v_gained <= 0 THEN
    v_bal := (SELECT room_cards FROM poker_profiles WHERE owner_id = v_uid);
    RETURN jsonb_build_object('ok', false,
      'reason', '再打 ' || (v_per - ((v_hands - v_base) % v_per)) || ' 手可领 1 张',
      'roomCards', v_bal);
  END IF;

  v_new_base := v_base + v_gained * v_per;
  UPDATE poker_profiles
     SET bonus_base = v_new_base,
         room_cards = room_cards + v_gained
   WHERE owner_id = v_uid
  RETURNING room_cards INTO v_bal;

  -- 流水
  INSERT INTO poker_card_grants (owner_id, kind, amount, balance_after, hand_base)
  VALUES (v_uid, 'hands', v_gained, v_bal, v_new_base);

  RETURN jsonb_build_object('ok', true, 'gained', v_gained, 'roomCards', v_bal);
END;
$$;
```

### 3.3 poker_consume_room_card(p_count int, p_token text) —— 建房扣卡（幂等）

```sql
CREATE OR REPLACE FUNCTION poker_consume_room_card(p_count int, p_token text)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_uid   uuid  := auth.uid();
  v_bal   int;
  v_count int  := COALESCE(p_count, 1);
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION '未登录' USING ERRCODE = '42501';
  END IF;
  IF v_count <= 0 THEN
    RAISE EXCEPTION '扣减数量无效' USING ERRCODE = '23514';
  END IF;

  -- 幂等：同 token 已扣过 → 直接返回当前余额（不二次扣）
  IF p_token IS NOT NULL AND EXISTS (
    SELECT 1 FROM poker_card_grants
    WHERE owner_id = v_uid AND token = p_token
  ) THEN
    v_bal := (SELECT room_cards FROM poker_profiles WHERE owner_id = v_uid);
    RETURN jsonb_build_object('ok', true, 'roomCards', v_bal,
      'reason', '已扣减（幂等命中）');
  END IF;

  -- 余额校验 + 扣减同事务
  SELECT room_cards INTO v_bal FROM poker_profiles WHERE owner_id = v_uid FOR UPDATE;
  IF v_bal IS NULL THEN
    RAISE EXCEPTION '档案不存在' USING ERRCODE = 'P0002';
  END IF;
  IF v_bal < v_count THEN
    RETURN jsonb_build_object('ok', false, 'reason', '房卡不足', 'roomCards', v_bal);
  END IF;

  UPDATE poker_profiles
     SET room_cards = room_cards - v_count
   WHERE owner_id = v_uid
  RETURNING room_cards INTO v_bal;

  -- 流水（token 列 = 幂等凭据，唯一约束 (owner_id, token WHERE token NOT NULL)）
  INSERT INTO poker_card_grants (owner_id, kind, amount, balance_after, token)
  VALUES (v_uid, 'consume', -v_count, v_bal, p_token);

  RETURN jsonb_build_object('ok', true, 'roomCards', v_bal);
END;
$$;
```

### 3.4 poker_record_hand(p_row jsonb) —— 战绩上报（owner_id 服务端填）

```sql
CREATE OR REPLACE FUNCTION poker_record_hand(p_row jsonb)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_uid   uuid  := auth.uid();
  v_won   int   := CASE WHEN (p_row->>'won')::boolean THEN 1 ELSE 0 END;
  v_net   int   := COALESCE((p_row->>'net')::int, 0);
  v_row   poker_records;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION '未登录' USING ERRCODE = '42501';
  END IF;

  -- 插入战绩流水（owner_id 由服务端 auth.uid() 填，p_row 不含 owner_id）
  INSERT INTO poker_records (
    owner_id, table_id, hand_no, seat, hole_cards, board, hand_type, net, won
  ) VALUES (
    v_uid,
    COALESCE(p_row->>'tableId', 'ai'),
    COALESCE((p_row->>'handNo')::int, 1),
    COALESCE((p_row->>'seat')::int, 0),
    COALESCE(p_row->>'hole', ''),
    COALESCE(p_row->>'board', ''),
    COALESCE(p_row->>'handType', ''),
    v_net,
    (p_row->>'won')::boolean
  )
  RETURNING * INTO v_row;

  -- 同函数累计档案统计（列权限收口后客户端不能再 update poker_profiles）
  UPDATE poker_profiles
     SET total_hands = total_hands + 1,
         total_wins  = total_wins  + v_won,
         total_net   = total_net   + v_net
   WHERE owner_id = v_uid;

  RETURN to_jsonb(v_row);
END;
$$;
```

## 真云验真 checklist（用户在 WorkBuddy 控制台执行）

以下每项需用户在 WorkBuddy 控制台手工验真，确认后勾选。本仓库无 CI/迁移
工具，无法自动执行。

- [ ] 1. `poker_card_grants` 建表成功（含 token 列 + 两个唯一索引）。
  - `SELECT column_name FROM information_schema.columns WHERE table_name = 'poker_card_grants';`
  - 应返回 id / owner_id / kind / amount / balance_after / day / hand_base / token / created_at。
- [ ] 2. 列权限收口生效：用客户端会话（anon/authenticated）直接 `UPDATE poker_profiles SET room_cards = 999` 应被拒（42501/权限拒绝）。
  - 直接 `INSERT INTO poker_card_grants ...` 也应被拒（REVOKE INSERT 生效）。
- [ ] 3. `poker_claim_daily` 函数创建成功：
  - 首次调用 `SELECT * FROM rpc('poker_claim_daily', '{"p_day":"2026-10-16"}');` 返回 `{ok:true, gained:2, roomCards:N}`。
  - 同日再调返回 `{ok:false, reason:'今日已领取，明天再来', roomCards:N}`（幂等）。
  - `poker_card_grants` 多一条 `kind='daily'` 流水，`balance_after` 与 profiles.room_cards 一致。
- [ ] 4. `poker_claim_hands` 函数创建成功：
  - 无可领时返回 `{ok:false, reason:'再打 N 手...'}`。
  - 可领时返回 `{ok:true, gained:N, roomCards:N}`，bonus_base 推进正确。
  - 并发两次调用只有一方 gained>0（FOR UPDATE 抢占生效）。
- [ ] 5. `poker_consume_room_card` 函数创建成功：
  - 余额不足返回 `{ok:false, reason:'房卡不足'}`。
  - 扣减成功返回 `{ok:true, roomCards:N}`，流水 `kind='consume', amount=-1`。
  - 同 token 重发返回 `{ok:true, reason:'已扣减（幂等命中）'}`，余额不二次扣。
- [ ] 6. `poker_record_hand` 函数创建成功：
  - 传入 `p_row`（不含 owner_id）返回插入的 record 行（含 owner_id = 当前用户）。
  - `poker_profiles.total_hands/total_wins/total_net` 同步累加正确。
  - 客户端直接 `UPDATE poker_profiles SET total_hands=999` 应被拒（列权限收口生效）。
- [ ] 7. 端到端：小程序端 `auth.claimDailyCards()` / `claimHandBonus()` /
